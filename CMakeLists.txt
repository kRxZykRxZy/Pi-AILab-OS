cmake_minimum_required(VERSION 3.16)
project(piai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_executable(piai src/main.cpp src/gguf_v3.cpp src/quant.cpp src/transformer.cpp src/api.cpp src/runtime.cpp src/platform.cpp src/inference_v4.cpp src/architecture_v2.cpp)
target_include_directories(piai PRIVATE include)
target_compile_options(piai PRIVATE -Wall -Wextra -Wpedantic -O3 -ffunction-sections -fdata-sections)
target_link_options(piai PRIVATE -Wl,--gc-sections)
if(UNIX)
 target_link_libraries(piai PRIVATE pthread)
endif()
install(TARGETS piai RUNTIME DESTINATION bin)
