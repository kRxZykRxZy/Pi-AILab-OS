cmake_minimum_required(VERSION 3.16)
project(piai VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -ffunction-sections -fdata-sections)
add_executable(piai
 src/main.cpp src/runtime.cpp src/config.cpp src/http.cpp src/system.cpp)
target_include_directories(piai PRIVATE include)
target_link_libraries(piai PRIVATE pthread)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_link_options(piai PRIVATE -Wl,--gc-sections)
endif()
install(TARGETS piai RUNTIME DESTINATION bin)
install(FILES packaging/piai.service DESTINATION lib/systemd/system)
